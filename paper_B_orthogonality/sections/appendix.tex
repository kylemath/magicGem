\section{Computational methods}
\label{sec:appendix}

All computations were run on an Apple M4 Max under macOS with Python~3.9 (NumPy, SciPy, SymPy) and C compiled with Apple clang; uniformly random arrangements were generated by Fisher--Yates shuffles.
The scripts and data are in \texttt{anc/}, with a \texttt{README} listing the commands that regenerate every table and figure.

\paragraph{The permutation threshold.}
\texttt{anc/kstar\_verify.py} checks in exact integer arithmetic that each witness arrangement ($5\le n\le40$) is a permutation of $1,\dots,N$ with all row sums $M(n)$, and computes the order of vanishing of $P_\gamma$ at $1$ from the binomial moments $\sum_j\binom jr\gamma_j$.
For the upper bounds it reduces the basis $\{t^i(t-1)^{k+1}\}$ of $\Lambda_{n,k}$ by LLL (exact integer arithmetic) and enumerates all lattice vectors with $\|\gamma\|_2^2\le n^5(n^2-1)/12$ by the Fincke--Pohst method with exact rational Gram--Schmidt coefficients, testing \eqref{eq:col_feasible} for every $p$.
Since $\Lambda_{n,k+1}\subset\Lambda_{n,k}$, feasibility is monotone in $k$, and the least level without a feasible vector is the upper bound.
The same script checks the rank-$2$ lattices $\Lambda_{n,n-3}$ for $14\le n\le400$.
The witnesses were found with \texttt{anc/realise.py} and \texttt{anc/anneal2.c}, starting from the squares of \Cref{lem:twin_squares} (\texttt{anc/constructions.py}; an independent implementation of the construction is \texttt{anc/twin\_squares.py}).

\paragraph{Variances.}
\texttt{anc/variance\_proof.py} derives \eqref{eq:var_identity} symbolically by M\"obius inversion over the $15$ set partitions of $\{1,2,3,4\}$, checks it by enumerating all permutations for random matrices with $N\le7$, computes the invariants \eqref{eq:inv_low}--\eqref{eq:inv_full} symbolically and checks them against the explicit matrices for $3\le n\le25$, and compares the result with the enumeration of all $9!$ arrangements for $n=3$.
Monte Carlo with $10^8$ arrangements per order ($3\le n\le8$) agrees with the exact means and standard deviations to within $5\times10^{-4}$ relative error (\texttt{anc/mc\_results.json}).
The densities in \Cref{fig:random_energy} were computed by numerical inversion of the characteristic function \cite{imhof1961}.

\paragraph{Meet-in-the-middle enumeration of $L_4$.}
The $16$ cells are split into the top two rows $A$ and the bottom two rows $B$.
For each of the $\binom{16}{8}=12{,}870$ value sets $V$ and each of the $8!$ bijections $V\to A$ the partial vector $(U_1,\dots,U_4)|_A$ is packed into an integer key; for the complementary set and each bijection $V^c\to B$ the key of the target minus the partial vector is formed; the two key lists are sorted and merged, and the sum over matching keys of the products of multiplicities is $|L_4|$.
This touches about $1.04\times10^9$ partial vectors instead of $16!\approx2.1\times10^{13}$ arrangements (\texttt{anc/lowmode\_enum.c}).

\texttt{anc/verify\_numbers.py} and \texttt{anc/balance\_check.py} recompute the remaining numbers (the former in exact arithmetic).

\section{Squares with many twin rows}
\label{sec:app_twin}

\begin{proof}[Proof of \Cref{lem:twin_squares}(1)]
Give the value $v$ the weight $|2v-(N+1)|$, an odd number in $\{1,3,\dots,N-1\}$; each weight $w$ belongs to exactly the two complementary values $(N+1\pm w)/2$.
Index the weights by $x=(w-1)/2\in\{0,\dots,2m^2-1\}$, $w_x=2x+1$.
It suffices to partition the indices into $m$ sets $R_0,\dots,R_{m-1}$ of size $n$ and choose signs $\sigma_x\in\{\pm1\}$ with $\sum_{x\in R_r}\sigma_xw_x=0$ for every $r$: then row $r$ receives the values $(N+1+\sigma_xw_x)/2$, $x\in R_r$, in any order, and row $m+r$ their complements in the same columns.
Every value occurs once, row $r$ sums to $n\frac{N+1}2=M(n)$, row $m+r$ to $n(N+1)-M(n)=M(n)$, and every column to $m(N+1)=M(n)$.
A \emph{domino} $\{x,x+1\}$ with signs $(+,-)$ or $(-,+)$ contributes $\mp2$.
If $m$ is even, give each row $m$ of the dominoes $\{2t,2t+1\}$ with alternating sign patterns.
If $m\ge7$ is odd, give row $r$ the four indices $4r,\dots,4r+3$ with signs $(+,+,+,-)$, contributing $(8r+1)+(8r+3)+(8r+5)-(8r+7)=16r+2$; the two indices $\alpha_r=5m-1-r$ and $\beta_r=5m+7r$ with signs $(+,-)$, contributing $2(\alpha_r-\beta_r)=-16r-2$; and $m-3$ dominoes with alternating sign patterns.
The quadruples fill $[0,4m-1]$ and the $\alpha_r$ fill $[4m,5m-1]$; consecutive $\beta_r$ leave gaps of exactly $6$ indices, and the remaining interval $[12m-6,\,2m^2-1]$ has even length $2m^2-12m+6\ge0$ because $m\ge7$.
So the unused indices form $3(m-1)+(m^2-6m+3)=m(m-3)$ dominoes, as required.
\end{proof}

\begin{proof}[Proof of \Cref{lem:twin_squares}(2)]
Start with a $7\times7$ magic square and border it successively to the orders $9,11,\dots,n$ by the following step, which turns a semi-magic square $T$ of order $n-2$ into one of order $n=2m+1$ ($m\ge4$) whose top and bottom rows are twins on the inner columns.
Let $N=n^2$, $c=(N+1)/2$, $A=4m$, $B=4m-2$.
The inner block is $T+(2n-2)$, which uses the values $2n-1,\dots,N-2n+2$; each inner row and column then sums to $M(n-2)+(n-2)(2n-2)=(n-2)(N+1)/2$.
The border uses the $4m=2n-2$ complementary pairs $\{p,N+1-p\}$, $1\le p\le4m$.
Put $s_{00}=N+1-A$, $s_{n-1,n-1}=A$, $s_{0,n-1}=B$, $s_{n-1,0}=N+1-B$.
Let $P=[2m,4m]\setminus\{A,B\}$ and $Q=[1,2m-1]$, both of size $2m-1$, so $\sum P-\sum Q=4m^2-4m+2$.
Choose an $m$-subset $T_+$ of $[2m,4m-3]\subset P$ and an $(m-1)$-subset $T_-$ of $Q$ with $\sum T_+-\sum T_-=c-A+B=2m^2+2m-1$.
This is possible: the $k$-subsets of an interval of integers have as sums all integers between the smallest and the largest such sum (raise one element by $1$ at a time), so $\sum T_+-\sum T_-$ takes every integer in $[m^2+m,\,3m^2-2m]$, which contains $2m^2+2m-1$ for $m\ge4$.
The inner top row consists of the values $N+1-p$ ($p\in T_+$) and $p$ ($p\in T_-$); the inner bottom row of their complements in the same columns; the inner left column of $N+1-p$ ($p\in P\setminus T_+$) and $p$ ($p\in Q\setminus T_-$); and the inner right column of their complements in the same rows.
The top row sums to $(m+1)(N+1)-A+B-\sum T_++\sum T_-=(2m+1)c=M(n)$, and the left column to $(m+1)(N+1)-A-B-\sum(P\setminus T_+)+\sum(Q\setminus T_-)=M(n)$, because $\sum(P\setminus T_+)-\sum(Q\setminus T_-)=4m^2-4m+2-(2m^2+2m-1)=c-A-B$.
The bottom row and right column then sum to $n(N+1)-M(n)=M(n)$, each inner row and column to $N+1+(n-2)(N+1)/2=M(n)$, and every value $1,\dots,N$ occurs once.
Bordering shifts the inner square by a constant $s$, and $(a+s)+(b+s)=N+1$ whenever $a+b$ equals the previous $N+1$, so earlier twin relations persist; this gives the twin rows $\ell,n-1-\ell$ on the columns $\ell+1,\dots,n-2-\ell$ for $0\le\ell\le(n-9)/2$.
\end{proof}

\section{Derivation of the variance formulas}
\label{sec:app_variance}

\paragraph{Joint moments.}
Let $b$ be a vector with $\sum_kb_k=0$, $p_r=\sum_kb_k^r$, and $z$ a uniformly random permutation of $b$.
For distinct cells, $\E[z_a^4]=p_4/N$, and with $(N)_k=N(N-1)\cdots(N-k+1)$,
\[
\E[z_a^3z_b]=\frac{-p_4}{(N)_2},\quad \E[z_a^2z_b^2]=\frac{p_2^2-p_4}{(N)_2},\quad \E[z_a^2z_bz_c]=\frac{2p_4-p_2^2}{(N)_3},\quad \E[z_az_bz_cz_d]=\frac{3p_2^2-6p_4}{(N)_4},
\]
by M\"obius inversion of the power sums over set partitions.
Write $m_\pi$ for $\E[z_az_bz_cz_d]$ when the equality pattern of $(a,b,c,d)$ is exactly the partition $\pi$ of the positions $\{1,2,3,4\}$.

\paragraph{Proof of \Cref{thm:var_identity}.}
The mean: $\E[z^{\mathsf T}Az]=\frac{p_2}N\tr A-\frac{p_2}{N(N-1)}\sum_{a\ne b}A_{ab}$ and $\sum_{a\ne b}A_{ab}=-\tr A$.
For the second moment let $S^{=}(\pi)$ be the sum of $A_{ab}A_{cd}$ over the tuples with pattern exactly $\pi$ and $S^{\ge}(\sigma)=\sum_{\pi\ge\sigma}S^{=}(\pi)$ the sum over tuples satisfying at least the equalities of $\sigma$; $S^{\ge}(\sigma)$ is an unrestricted sum.
M\"obius inversion gives $\E[(z^{\mathsf T}Az)^2]=\sum_\pi m_\pi S^{=}(\pi)=\sum_\sigma c(\sigma)S^{\ge}(\sigma)$ with $c(\sigma)=\sum_{\pi\le\sigma}\mu(\pi,\sigma)m_\pi$.
If $\sigma$ has a singleton block, the corresponding index occurs in a single factor of $A_{ab}A_{cd}$ and summing over it gives a row sum, so $S^{\ge}(\sigma)=0$.
The remaining partitions are $12|34$, $13|24$, $14|23$ and $1234$, with $S^{\ge}$ equal to $(\tr A)^2$, $\tr A^2$, $\tr A^2$, $\sum_aA_{aa}^2$, and
\[
\begin{aligned}c(12|34)=c(13|24)=c(14|23)&=m_{22}-2m_{211}+m_{1111},\\ c(1234)&=m_4-4m_{31}-3m_{22}+12m_{211}-6m_{1111}.\end{aligned}
\]
For the centred values $1,\dots,N$, $p_2=N(N^2-1)/12$ and $p_4=N(N^2-1)(3N^2-7)/240$, and these coefficients simplify to $N(N^2-1)(5N+6)/720$ and $-N^2(N+1)^2/120$; subtracting $\bigl(\frac{N(N+1)}{12}\tr A\bigr)^2$ gives \eqref{eq:var_identity}.

\paragraph{The invariants.}
With $\Gamma_{kl}=\Cov(\varphi_k,\varphi_l)$ and $h_c=\sum_ka_{k,c}^2$,
\[
\tr A=\frac1N\tr\Gamma,\qquad\tr A^2=\frac1{N^2}\sum_{k,l}\Gamma_{kl}^2,\qquad\sum_aA_{aa}^2=\sum_ch_c^2 .
\]
For a line, $(\mathbf 1_L(c)-\frac1n)^2=\frac1{n^2}+(1-\frac2n)\mathbf 1_L(c)$.
Hence $N^2h_c=x_c^2+y_c^2+\frac2{n^2}+(1-\frac2n)(d_{\mathrm m}+d_{\mathrm a})(c)$ for $\Elow$ and $N^2h_c=\frac{2n+2}{n^2}+(1-\frac2n)(2+d_{\mathrm m}(c)+d_{\mathrm a}(c))$ for $\Efull$, where $d_{\mathrm m},d_{\mathrm a}$ are the diagonal indicators.
The entries of $\Gamma$ are given in the proof of \Cref{prop:random_energy_moments}.
The sums needed are $\sum_cx_c^2=\frac{n^2(n^2-1)}{12}$, $\sum_cx_c^4=\frac{n^2(n^2-1)(3n^2-7)}{240}$, $\sum_cx_c^2y_c^2=\bigl(\frac{n(n^2-1)}{12}\bigr)^2$, $\sum_{c\in D}x_c^2=\sum_{c\in D}y_c^2=\frac{n(n^2-1)}{12}$ for either diagonal $D$, $\sum_c(d_{\mathrm m}+d_{\mathrm a})=2n$ and $\sum_c(d_{\mathrm m}+d_{\mathrm a})^2=2n+2\varepsilon$, where $\varepsilon=1$ for odd and $0$ for even $n$ is the number of cells on both diagonals.
Collecting terms gives the following invariants; where two values are given, the first is for odd and the second for even $n$.
For $\Elow$:
\begin{equation}
\begin{aligned}
\tr A&=\frac{(n-1)(n^3+n^2+12)}{6n^4},\\
\tr A^2&=\frac{(n-1)^2(n^6+2n^5+n^4+144)}{72n^8},\qquad\frac{n^8-2n^6+n^4+144n^2-288n+288}{72n^8},\\
\sum_aA_{aa}^2&=\frac{(n-1)(7n^7+7n^6-13n^5+107n^4+240n^2-720n+720)}{180n^{10}},\\
&\qquad\frac{7n^8-20n^6+120n^5-107n^4+240n^3-1320n^2+2880n-2160}{180n^{10}}.
\end{aligned}
\label{eq:inv_low}
\end{equation}
For $\Efull$:
\begin{equation}
\begin{aligned}
\tr A&=\frac{2(n^2-1)}{n^4},\qquad
\tr A^2=\frac{2(n-1)(n^2+n-1)}{n^8},\ \ \frac{2(n^3-2n+2)}{n^8},\\
\sum_aA_{aa}^2&=\frac{2(n-1)(2n-1)(n^2+2n-2)}{n^{10}},\ \ \frac{2(2n^4+n^3-10n^2+12n-6)}{n^{10}}.
\end{aligned}
\label{eq:inv_full}
\end{equation}

\section{A witness for \texorpdfstring{$k^*(5)=4$}{k*(5)=4}}
\label{sec:app_witnesses}

The arrangement
\[
\begin{pmatrix}1&22&11&24&7\\23&3&12&2&25\\18&6&19&17&5\\4&21&14&10&16\\20&9&15&8&13\end{pmatrix}
\]
has all row sums $M(5)=65$ and column sums $66,61,71,61,66$, i.e.\ $\gamma=(1,-4,6,-4,1)$ and $P_\gamma=(1-t)^4$; it satisfies the degree-$3$ conditions without being semi-magic.
The witnesses for $6\le n\le40$ are in \texttt{anc/data\_witnesses\_variances.json} ($n\le13$) and \texttt{anc/kstar\_witnesses.json} ($14\le n\le40$).
